from app import db
from app.models.chytapay import ChytaPayToken
from datetime import datetime, timedelta

# 🔑 NUEVOS TOKENS (reemplaza con los que te dieron)
NUEVO_ID_TOKEN = "eyJraWQiOiJ......"
NUEVO_REFRESH_TOKEN = "eyJraWQiOiJ......."

# Actualizar el token existente
token = ChytaPayToken.query.first()
if token:
    token.access_token = NUEVO_ID_TOKEN
    token.refresh_token = NUEVO_REFRESH_TOKEN
    token.expires_at = datetime.utcnow() + timedelta(hours=1)  # 1 hora de validez
    print("✅ Token actualizado")
else:
    # Si no existe, crear uno nuevo
    token = ChytaPayToken(
        access_token=NUEVO_ID_TOKEN,
        refresh_token=NUEVO_REFRESH_TOKEN,
        expires_at=datetime.utcnow() + timedelta(hours=1)
    )
    db.session.add(token)
    print("✅ Token creado")

db.session.commit()
print(f"✅ Tokens guardados. Expira: {token.expires_at}")
